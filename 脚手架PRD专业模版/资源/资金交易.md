<script>
    (function() {
        // 确保 Mermaid 初始化配置兼容且安全
        mermaid.initialize({
            startOnLoad: true,
            theme: 'base',
            themeVariables: {
                'background': '#ffffff',
                'primaryColor': '#e1f5fe',
                'primaryBorderColor': '#01579b',
                'lineColor': '#5f7f9e',
                'secondaryColor': '#fff3e0',
                'tertiaryColor': '#e8f5e9',
                'clusterBkg': '#fafcfd',
                'clusterBorder': '#b0c4de',
                'fontFamily': 'Segoe UI, Roboto, "Noto Sans", system-ui, sans-serif'
            },
            flowchart: {
                useMaxWidth: true,
                htmlLabels: true,
                curve: 'basis',
                padding: 20,
                nodeSpacing: 55,
                rankSpacing: 80
            },
            securityLevel: 'loose'
        });

        // 为三个子图增加明显的背景色（提升视觉区分）
        function applySubgraphStyles() {
            setTimeout(() => {
                const svg = document.querySelector('.mermaid svg');
                if (!svg) return;
                const clusters = svg.querySelectorAll('.cluster');
                for (let cluster of clusters) {
                    const labelElem = cluster.querySelector('.label');
                    if (!labelElem) continue;
                    const text = labelElem.textContent || '';
                    const rect = cluster.querySelector('rect');
                    if (!rect) continue;
                    if (text.includes('工程仓业务系统')) {
                        rect.setAttribute('fill', '#eef6ff');
                        rect.setAttribute('stroke', '#1e88e5');
                    } else if (text.includes('支付中台')) {
                        rect.setAttribute('fill', '#fff7e6');
                        rect.setAttribute('stroke', '#f39c12');
                    } else if (text.includes('对账系统')) {
                        rect.setAttribute('fill', '#edf7ed');
                        rect.setAttribute('stroke', '#2e7d32');
                    }
                }
            }, 150);
        }

        window.addEventListener('load', () => {
            try {
                const mermaidDiv = document.querySelector('.mermaid');
                if (mermaidDiv && !mermaidDiv.getAttribute('data-processed')) {
                    mermaid.contentLoaded();
                }
                // 监听渲染完成
                const observer = new MutationObserver(() => applySubgraphStyles());
                observer.observe(document.querySelector('.mermaid-wrapper'), { childList: true, subtree: true });
                applySubgraphStyles();
            } catch (e) {
                console.warn("辅助样式应用", e);
            }
        });
    })();
</script>
